                    onChange={(e) => setReadings({ ...readings, battery: Number(e.target.value) })}
                    className="w-full text-center bg-transparent text-[13px] font-mono font-black text-cyan-300 outline-none"
                  />
                  <span className="text-[7px] font-mono text-slate-500">VDC Float</span>
                </div>

                <div className="p-2 rounded-[8px] bg-[#090d16] border border-slate-800 text-center">
                  <span className="text-[7px] font-mono uppercase text-slate-400 block">Temp</span>
                  <input
                    type="number"
                    value={readings.temperature}
                    onChange={(e) => setReadings({ ...readings, temperature: Number(e.target.value) })}
                    className="w-full text-center bg-transparent text-[13px] font-mono font-black text-amber-300 outline-none"
                  />
                  <span className="text-[7px] font-mono text-slate-500">°C Water Jacket</span>
                </div>

                <div className="p-2 rounded-[8px] bg-[#090d16] border border-slate-800 text-center">
                  <span className="text-[7px] font-mono uppercase text-slate-400 block">Lube Press</span>
                  <input
                    type="number"
                    step="0.1"
                    value={readings.pressure}
                    onChange={(e) => setReadings({ ...readings, pressure: Number(e.target.value) })}
                    className="w-full text-center bg-transparent text-[13px] font-mono font-black text-cyan-300 outline-none"
                  />
                  <span className="text-[7px] font-mono text-slate-500">Bar Gauge</span>
                </div>

                <div className="p-2 rounded-[8px] bg-[#090d16] border border-slate-800 text-center">
                  <span className="text-[7px] font-mono uppercase text-slate-400 block">Call Nature</span>
                  <select
                    value={callType}
                    onChange={(e) => setCallType(e.target.value)}
                    className="w-full text-center bg-transparent text-[9px] font-bold text-cyan-300 outline-none"
                  >
                    <option value="Breakdown" className="bg-[#0f172a]">Breakdown</option>
                    <option value="PM" className="bg-[#0f172a]">PM</option>
                    <option value="Inspection" className="bg-[#0f172a]">Inspection</option>
                  </select>
                  <span className="text-[7px] font-mono text-slate-500">Dispatch Code</span>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* --- TAB 2: DIAGNOSTIC LOGS & SPARES --- */}
        {activeTab === 'diagnostics' && (
          <div className="space-y-4 animate-in fade-in duration-150">
            {/* Quick Parts Spares Tray */}
            <div className="p-3.5 rounded-[12px] bg-[#111827] border border-slate-800 space-y-2">
              <span className="text-[8px] font-mono font-bold uppercase tracking-wider text-cyan-400 block">
                Quick Spares Picker
              </span>
              <div className="flex flex-wrap gap-1.5">
                {QUICK_PARTS.map((item, idx) => (
                  <button
                    key={idx}
                    type="button"
                    onClick={() => addQuickPart(item)}
                    className="px-2 py-1 rounded-[6px] text-[8px] font-mono bg-[#090d16] border border-slate-700 hover:border-cyan-400 text-slate-300 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Plus /> {item.name}
                  </button>
                ))}
              </div>
            </div>

            {/* Allocated Materials List */}
            <div className="p-3.5 rounded-[12px] bg-[#111827] border border-slate-800 space-y-2">
              <div className="flex items-center justify-between">
                <span className="text-[8px] font-mono font-bold uppercase tracking-wider text-slate-300">
                  Allocated Consumables & Spares
                </span>
                <span className="text-[8px] font-mono text-cyan-400">{parts.length} line items</span>
              </div>

              <div className="divide-y divide-slate-800/80">
                {parts.map((p) => (
                  <div key={p.id} className="py-1.5 flex items-center justify-between">
                    <div>
                      <p className="text-[10px] font-bold text-white">{p.name}</p>
                      <span className="text-[7px] font-mono text-slate-400">{p.unit}</span>
                    </div>
                    <div className="flex items-center gap-2">
                      <span className="px-2 py-0.5 rounded-[4px] bg-[#090d16] text-[10px] font-mono font-bold text-cyan-300 border border-slate-800">
                        ×{p.qty}
                      </span>
                      <button
                        type="button"
                        onClick={() => removePart(p.id)}
                        className="p-1 text-slate-500 hover:text-red-400 transition"
                      >
                        <Icons.Trash />
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            </div>

            {/* Diagnostic Log Fields */}
            <div className="space-y-2.5">
              <div className="p-3 rounded-[10px] bg-[#111827] border border-slate-800">
                <label className="block text-[7px] font-mono uppercase text-slate-400 mb-1">
                  1. Problem Reported by Facility
                </label>
                <textarea
                  rows={2}
                  value={problemReported}
                  onChange={(e) => setProblemReported(e.target.value)}
                  className="w-full bg-[#090d16] p-2 rounded-[6px] text-[10px] text-slate-200 border border-slate-800 focus:border-cyan-500 outline-none resize-none"
                />
              </div>

              <div className="p-3 rounded-[10px] bg-[#111827] border border-slate-800">
                <label className="block text-[7px] font-mono uppercase text-slate-400 mb-1">
                  2. Root Cause & Defects Found
                </label>
                <textarea
                  rows={2}
                  value={rootCause}
                  onChange={(e) => setRootCause(e.target.value)}
                  className="w-full bg-[#090d16] p-2 rounded-[6px] text-[10px] text-slate-200 border border-slate-800 focus:border-cyan-500 outline-none resize-none"
                />
              </div>

              <div className="p-3 rounded-[10px] bg-[#111827] border border-slate-800">
                <label className="block text-[7px] font-mono uppercase text-cyan-400 font-bold mb-1">
                  3. Corrective Service Rendered
                </label>
                <textarea
                  rows={3}
                  value={correctiveAction}
                  onChange={(e) => setCorrectiveAction(e.target.value)}
                  className="w-full bg-[#090d16] p-2 rounded-[6px] text-[10px] text-slate-200 border border-slate-800 focus:border-cyan-500 outline-none resize-none"
                />
              </div>

              <div className="p-3 rounded-[10px] bg-[#111827] border border-slate-800">
                <label className="block text-[7px] font-mono uppercase text-slate-400 mb-1">
                  4. Engineer Recommendations
                </label>
                <textarea
                  rows={2}
                  value={preventiveAdvice}
                  onChange={(e) => setPreventiveAdvice(e.target.value)}
                  className="w-full bg-[#090d16] p-2 rounded-[6px] text-[10px] text-slate-200 border border-slate-800 focus:border-cyan-500 outline-none resize-none"
                />
              </div>
            </div>
          </div>
        )}

        {/* --- TAB 3: SIGNOFF & AUTHORIZATION --- */}
        {activeTab === 'signoff' && (
          <div className="space-y-4 animate-in fade-in duration-150">
            {/* Engineer Identity Certificate */}
            <div className="p-3.5 rounded-[12px] bg-[#111827] border border-slate-800 space-y-2">
              <span className="text-[8px] font-mono font-bold uppercase tracking-wider text-cyan-400 block">
                Certified Service Engineer
              </span>
              <div className="flex items-center justify-between p-2 rounded-[8px] bg-[#090d16] border border-slate-800">
                <div>
                  <p className="text-[11px] font-black text-white">{selectedSignatory.name}</p>
                  <span className="text-[8px] font-mono text-slate-400">{selectedSignatory.stamp}</span>
                </div>
                <button
                  type="button"
                  onClick={() => setSignatorySheetOpen(true)}
                  className="px-2 py-1 rounded-[6px] text-[8px] font-mono font-bold text-cyan-300 bg-cyan-950 border border-cyan-500/40"
                >
                  Change
                </button>
              </div>
            </div>

            {/* Client Plant Supervisor */}
            <div className="p-3.5 rounded-[12px] bg-[#111827] border border-slate-800 space-y-2">
              <label className="block text-[8px] font-mono font-bold uppercase tracking-wider text-slate-300">
                Client Plant Supervisor / Endorser
              </label>
              <input
                type="text"
                value={clientRepresentative}
                onChange={(e) => setClientRepresentative(e.target.value)}
                placeholder="Name & Title"
                className="w-full h-[36px] px-2.5 rounded-[8px] bg-[#090d16] text-[10px] font-bold text-slate-200 border border-slate-800 focus:border-cyan-500 outline-none"
              />
            </div>

            {/* Interactive Signature Touch Surface */}
            <div className="p-3.5 rounded-[12px] bg-[#111827] border border-slate-800 space-y-2">
              <div className="flex items-center justify-between">
                <span className="text-[8px] font-mono font-bold uppercase tracking-wider text-cyan-400">
                  Digital Signature Pad
                </span>
                <button
                  type="button"
                  onClick={clearCanvas}
                  className="text-[8px] font-mono text-red-400 flex items-center gap-1 hover:underline"
                >
                  <Icons.RotateCcw /> Clear Pad
                </button>
              </div>

              <div className="h-32 rounded-[8px] bg-white relative overflow-hidden border border-slate-700 touch-none">
                <canvas
                  ref={canvasRef}
                  width={380}
                  height={128}
                  onMouseDown={startDrawing}
                  onMouseMove={draw}
                  onMouseUp={stopDrawing}
                  onMouseLeave={stopDrawing}
                  onTouchStart={startDrawing}
                  onTouchMove={draw}
                  onTouchEnd={stopDrawing}
                  className="w-full h-full cursor-crosshair"
                />
                {!signatureDataUri && (
                  <div className="absolute inset-0 pointer-events-none flex items-center justify-center text-[10px] font-mono text-slate-400">
                    ✍ Sign directly with finger or stylus
                  </div>
                )}
              </div>
            </div>
          </div>
        )}

      </div>

      {/* --- FLOATING COMMAND CENTER FAB (V2 LIVE GLOW) --- */}
      <div className="fixed bottom-6 right-5 z-40">
        <button
          type="button"
          onClick={handleFinalSave}
          className="w-[54px] h-[54px] rounded-[18px] text-white flex items-center justify-center v2-fab-live active:scale-90 transition border border-cyan-400/40 bg-gradient-to-br from-cyan-600 via-cyan-800 to-slate-900 cursor-pointer"
          title="Certify & Save Dossier"
        >
          <Icons.Floppy />
        </button>
      </div>

      {/* --- BOTTOM SHEET: CLIENT DISPATCH PICKER --- */}
      {clientSheetOpen && (
        <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-xs flex items-end">
          <div className="w-full max-w-md mx-auto bg-[#0f172a] rounded-t-[22px] max-h-[75vh] flex flex-col p-4 space-y-3 border-t border-cyan-500/30 animate-in slide-in-from-bottom duration-150">
            <div className="w-8 h-1 rounded-full bg-slate-700 mx-auto" />
            <div className="flex items-center justify-between pb-1 border-b border-slate-800">
              <span className="text-[11px] font-mono font-bold uppercase text-white">Select Industrial Client</span>
              <button
                onClick={() => setClientSheetOpen(false)}
                className="text-slate-400 hover:text-white"
              >
                ✕
              </button>
            </div>

            <div className="relative">
              <input
                type="text"
                value={clientSearch}
                onChange={(e) => setClientSearch(e.target.value)}
                placeholder="Filter by facility name or location..."
                className="w-full h-[36px] pl-8 pr-3 rounded-[8px] bg-[#090d16] text-[10px] text-white border border-slate-800 focus:border-cyan-500 outline-none"
              />
              <div className="absolute left-2.5 top-1/2 -translate-y-1/2 text-slate-500">
                <Icons.Search />
              </div>
            </div>

            <div className="space-y-1.5 overflow-y-auto flex-1">
              {CLIENTS_DB.filter(c => c.name.toLowerCase().includes(clientSearch.toLowerCase())).map(c => (
                <div
                  key={c.id}
                  onClick={() => {
                    setSelectedClient(c);
                    setClientSheetOpen(false);
                    showToast(`Assigned ${c.name}`);
                  }}
                  className="p-2.5 rounded-[8px] bg-[#111827] hover:bg-[#1e293b] border border-slate-800 cursor-pointer flex items-center justify-between"
                >
                  <div>
                    <p className="text-[11px] font-black text-white">{c.name}</p>
                    <p className="text-[8px] text-cyan-400 font-mono mt-0.5">{c.plant} · {c.region}</p>
                  </div>
                  <span className="text-[8px] font-mono text-slate-400 uppercase">Select</span>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* --- BOTTOM SHEET: SIGNATORY ROSTER --- */}
      {signatorySheetOpen && (
        <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-xs flex items-end">
          <div className="w-full max-w-md mx-auto bg-[#0f172a] rounded-t-[22px] max-h-[75vh] flex flex-col p-4 space-y-3 border-t border-cyan-500/30 animate-in slide-in-from-bottom duration-150">
            <div className="w-8 h-1 rounded-full bg-slate-700 mx-auto" />
            <div className="flex items-center justify-between pb-1 border-b border-slate-800">
              <span className="text-[11px] font-mono font-bold uppercase text-white">Authorized Engineer Roster</span>
              <button
                onClick={() => setSignatorySheetOpen(false)}
                className="text-slate-400 hover:text-white"
              >
                ✕
              </button>
            </div>

            <div className="space-y-1.5 overflow-y-auto flex-1">
              {SIGNATORIES_DB.map(s => (
                <div
                  key={s.id}
                  onClick={() => {
                    setSelectedSignatory(s);
                    setSignatorySheetOpen(false);
                    showToast(`Assigned ${s.name}`);
                  }}
                  className={`p-2.5 rounded-[8px] bg-[#111827] hover:bg-[#1e293b] border cursor-pointer flex items-center justify-between ${
                    selectedSignatory.id === s.id ? 'border-cyan-500 text-cyan-300' : 'border-slate-800 text-white'
                  }`}
                >
                  <div>
                    <p className="text-[11px] font-bold">{s.name}</p>
                    <p className="text-[8px] font-mono text-slate-400">{s.stamp}</p>
                  </div>
                  {selectedSignatory.id === s.id && <Icons.CheckCircle />}
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* --- TELEPORT ERROR TOAST --- */}
      {toast && (
        <div
          className={`fixed bottom-20 left-1/2 -translate-x-1/2 z-50 text-[9px] font-bold px-3.5 py-2 rounded-[10px] shadow-2xl max-w-[88vw] flex items-center gap-2 animate-in fade-in slide-in-from-bottom-2 duration-150 ${
            toast.isError
              ? 'bg-red-950 text-red-200 border border-red-500/60 shadow-[0_0_20px_rgba(239,68,68,0.3)]'
              : 'bg-cyan-950 text-cyan-200 border border-cyan-500/60 shadow-[0_0_20px_rgba(6,182,212,0.3)]'
          }`}
        >
          {toast.isError && <Icons.AlertOctagon />}
          <span className="truncate">{toast.text}</span>
        </div>
      )}

    </div>
  );
}
