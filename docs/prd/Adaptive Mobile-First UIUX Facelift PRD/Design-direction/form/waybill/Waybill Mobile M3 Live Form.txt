export default function App() {
  // Form State
  const [docType, setDocType] = useState(null); // 'external' | 'internal' | null (triggers gateway)

  const [purpose, setPurpose] = useState('Supply');
  const [plate, setPlate] = useState('ABC-123');
  const [driverName, setDriverName] = useState('');

  // Custody Details
  const [deliveredBy, setDeliveredBy] = useState('');
  const [receivedBy, setReceivedBy] = useState('');
  const [location, setLocation] = useState('');

    make: false,
    partNo: false,
    condition: true
  });
  const [customColumns, setCustomColumns] = useState([]); // { key, label }

  // Line Items
  const [items, setItems] = useState([

      id: 'item-1',
      description: 'Industrial Air Filter Replacement Pack',
      quantity: 2,
      unit: 'Units',
      make: '',
      partNo: '',
      condition: 'good',
      custom_data: {}
    }
  ]);

  // Signatures
  const [signatures, setSignatures] = useState({
    sender: { captured: false, url: null, visible: true, name: 'John Doe' },
    receiver: { captured: false, url: null, visible: true, name: '' }
  });

  // Notes & Terms (collapsible)
  const [notesOpen, setNotesOpen] = useState(false);
  const [notesTitle, setNotesTitle] = useState('Notes');
  const [notesText, setNotesText] = useState('');
  const [termsOpen, setTermsOpen] = useState(false);
  const [showTerms, setShowTerms] = useState(false);
  const [termsText, setTermsText] = useState('1. Title and risk of goods pass to consignee upon physical receipt.\n2. Discrepancies must be endorsed on this delivery note within 24 hours.');

  // Overlays / Sheets
  const [clientOverlay, setClientOverlay] = useState(false);
  const [clientQuery, setClientQuery] = useState('');
  const [invoiceOverlay, setInvoiceOverlay] = useState(false);
  const [invoiceQuery, setInvoiceQuery] = useState('');
  const [importOverlay, setImportOverlay] = useState(false);
  const [importJson, setImportJson] = useState('');
  const [columnManagerOpen, setColumnManagerOpen] = useState(false);
  const [moreSettingsOpen, setMoreSettingsOpen] = useState(false);
  const [signatorySheetRole, setSignatorySheetRole] = useState(null);
  const [activeDrawRole, setActiveDrawRole] = useState(null);
  const [clearAlertOpen, setClearAlertOpen] = useState(false);
  const [invalidRowId, setInvalidRowId] = useState(null);

  // Notifications
  const [toastMessage, setToastMessage] = useState(null);

  const fileInputRef = useRef(null);
  const canvasRef = useRef(null);
  const isDrawing = useRef(false);
  const lastPos = useRef({ x: 0, y: 0 });

  const showToast = (msg) => {
    setToastMessage(msg);
    setTimeout(() => {
      setToastMessage(prev => (prev === msg ? null : prev));
    }, 2800);
  };

  // --- Gateway Flow ---
  const handleSelectGatewayType = (type) => {
    setDocType(type);
    if (type === 'external') {
      setWaybillNo('WBL-E-000001');
      setPurpose('Supply');
      setLocation(client ? client.sub : '');
    } else {
      setWaybillNo('WBL-I-000001');
      setPurpose('Transfer');
      setLocation('Inter-depot movement route');
    }
  };

  // --- Item Manipulations ---
  const createDefaultItem = (idx = items.length + 1) => ({
    id: `item-${Date.now()}-${idx}`,
    description: '',
    quantity: 1,
    unit: 'Units',
    make: '',
    partNo: '',
    condition: 'good',
    custom_data: {}
  });

  const addItem = (afterIdx = null) => {
    const newItem = createDefaultItem();
    if (afterIdx === null) {
      setItems([...items, newItem]);
    } else {
      const copy = [...items];
      copy.splice(afterIdx + 1, 0, newItem);
      setItems(copy);
    }
    showToast('Row added');
  };

  const updateItem = (index, field, value) => {
    const copy = [...items];
    copy[index][field] = value;
    setItems(copy);
  };

  const updateItemCustomData = (index, key, value) => {
    const copy = [...items];
    copy[index].custom_data = { ...copy[index].custom_data, [key]: value };
    setItems(copy);
  };

  const removeItem = (index) => {
    if (items.length <= 1) {
      setItems([createDefaultItem(1)]);
      showToast('Item reset to blank');
      return;
    }
    setItems(items.filter((_, i) => i !== index));
  };

  const duplicateItem = (index) => {
    const clone = {
      ...items[index],
      id: `item-${Date.now()}`,
      custom_data: { ...items[index].custom_data }
    };
    const copy = [...items];
    copy.splice(index + 1, 0, clone);
    setItems(copy);
    showToast('Item duplicated');
  };

  const moveItem = (index, dir) => {
    const target = index + dir;
    if (target < 0 || target >= items.length) return;
    const copy = [...items];
    const temp = copy[index];
    copy[index] = copy[target];
    copy[target] = temp;
    setItems(copy);
  };

  const handleClearAll = () => {
    setItems([createDefaultItem(1)]);
    setClearAlertOpen(false);
    showToast('All items cleared');
  };

  // --- Custom Column Management ---
  const handleAddCustomColumn = () => {
    if (customColumns.length >= 4) {
      showToast('Maximum 4 custom columns allowed.');
      return;
    }
    const label = prompt('Enter Column Header Name:');
    if (!label || !label.trim()) return;
    const key = `col_${Date.now()}`;
    setCustomColumns([...customColumns, { key, label: label.trim() }]);
    showToast(`Column "${label.trim()}" added`);
  };

  const handleRemoveCustomColumn = (key) => {
    setCustomColumns(customColumns.filter(c => c.key !== key));
  };

  // --- Canvas Drawing Logic ---
  useEffect(() => {
    if (!activeDrawRole || !canvasRef.current) return;
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    const rect = canvas.getBoundingClientRect();
    canvas.width = rect.width * 2;
    canvas.height = rect.height * 2;
    ctx.scale(2, 2);
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';
    ctx.lineWidth = 2.4;
    ctx.strokeStyle = '#0F172A';
  }, [activeDrawRole]);

  const getCanvasPos = (e) => {
    const canvas = canvasRef.current;
    const rect = canvas.getBoundingClientRect();
    const cx = e.touches ? e.touches[0].clientX : e.clientX;
    const cy = e.touches ? e.touches[0].clientY : e.clientY;
    return { x: cx - rect.left, y: cy - rect.top };
  };

  const startDraw = (e) => {
    isDrawing.current = true;
    lastPos.current = getCanvasPos(e);
  };

  const moveDraw = (e) => {
    if (!isDrawing.current || !canvasRef.current) return;
    e.preventDefault();
    const ctx = canvasRef.current.getContext('2d');
    const pos = getCanvasPos(e);
    ctx.beginPath();
    ctx.moveTo(lastPos.current.x, lastPos.current.y);
    ctx.lineTo(pos.x, pos.y);
    ctx.stroke();
    lastPos.current = pos;
  };

  const stopDraw = () => {
    isDrawing.current = false;
  };

  const resetCanvas = () => {
    if (!canvasRef.current) return;
    const ctx = canvasRef.current.getContext('2d');
    ctx.clearRect(0, 0, canvasRef.current.width, canvasRef.current.height);
  };

  const saveDrawing = () => {
    if (!canvasRef.current || !activeDrawRole) return;
    const url = canvasRef.current.toDataURL('image/png');
    setSignatures(prev => ({
      ...prev,
      [activeDrawRole]: { ...prev[activeDrawRole], captured: true, url }
    }));
    setActiveDrawRole(null);
    showToast('Signature drawing saved');
  };

  const handleUploadClick = (role) => {
    if (fileInputRef.current) {
      fileInputRef.current.onchange = (e) => {
        const file = e.target.files?.[0];
        if (file) {
          const reader = new FileReader();
          reader.onload = () => {
            setSignatures(prev => ({
              ...prev,
              [role]: { ...prev[role], captured: true, url: reader.result }
            }));
            showToast('Signature image uploaded');
          };
          reader.readAsDataURL(file);
        }
      };
      fileInputRef.current.click();
    }
  };

  const handlePickSignatory = (person) => {
    if (!signatorySheetRole) return;
    const canvas = document.createElement('canvas');
    canvas.width = 260;
    canvas.height = 80;
    const ctx = canvas.getContext('2d');
    ctx.font = 'bold 16px sans-serif';
    ctx.fillStyle = '#1e3a8a';
    ctx.fillText(person.name, 12, 38);
    ctx.font = '10px monospace';
    ctx.fillStyle = '#64748b';
    ctx.fillText(person.role.toUpperCase(), 12, 58);

    setSignatures(prev => ({
      ...prev,
      [signatorySheetRole]: {
        ...prev[signatorySheetRole],
        captured: true,
        url: canvas.toDataURL('image/png'),
        name: person.name
      }
    }));
    setSignatorySheetRole(null);
    showToast(`Signatory ${person.name} attached`);
  };

  // --- Save Validation ---
  const handleSave = (statusLabel = 'dispatched') => {
    if (!waybillNo.trim()) {
      showToast('Waybill number is missing or invalid.');
      return;
    }
    if (docType === 'external' && !client) {
      showToast('Client account must be selected for external waybills.');
      return;
    }
    if (docType === 'internal' && !receivedBy.trim()) {
      showToast('Recipient name is required for internal waybills.');
      return;
    }
    if (!date) {
      showToast('Date is required.');
      return;
    }
    if (items.length === 0) {
      showToast('Line items list cannot be empty.');
      return;
    }

    for (let i = 0; i < items.length; i++) {
      if (!items[i].description.trim() || Number(items[i].quantity) <= 0) {
        setInvalidRowId(items[i].id);
        setTimeout(() => setInvalidRowId(null), 2500);
        showToast(`Item ${i + 1} is missing a description or has quantity ≤ 0.`);
        return;
      }
    }

    showToast(`Waybill ${waybillNo} successfully saved (${statusLabel})!`);
  };

  // --- JSON Import Apply ---
  const handleApplyImport = () => {
    try {
      const parsed = JSON.parse(importJson);
      if (parsed.items && Array.isArray(parsed.items)) {
        setItems(parsed.items.map((it, idx) => ({ ...createDefaultItem(idx), ...it })));
      }
      if (parsed.waybillNo) setWaybillNo(parsed.waybillNo);
      if (parsed.driverName) setDriverName(parsed.driverName);
      if (parsed.plate) setPlate(parsed.plate);
      if (parsed.location) setLocation(parsed.location);
      setImportOverlay(false);
      showToast('Waybill payload imported');
    } catch {
      showToast('Import Failed: Invalid JSON schema');
    }
  };

  // ==========================================
  // VIEW 1: INITIAL GATEWAY OVERLAY (Create Mode)
  // ==========================================
  if (docType === null) {
    return (
      <div className="min-h-screen bg-slate-900/90 flex items-center justify-center p-3 font-sans antialiased selection:bg-indigo-100">
        <div className="w-full max-w-sm bg-white rounded-3xl p-5 shadow-2xl space-y-4 border border-slate-100">
          <div>
            <div className="text-[10px] font-bold uppercase tracking-widest text-slate-400 font-mono">
              Create Document
            </div>
            <h1 className="text-2xl font-black text-slate-900 mt-1">
              New <span className="text-indigo-600">Waybill</span>
            </h1>
            <p className="text-xs text-slate-500 mt-0.5">Select document type</p>
          </div>

          <div className="space-y-2.5">
            <button
              type="button"
              onClick={() => handleSelectGatewayType('external')}
              className="w-full p-3.5 rounded-2xl border border-slate-200 hover:border-indigo-600 hover:bg-indigo-50/50 text-left transition active:scale-[0.98]"
            >
              <div className="flex items-center justify-between">
                <span className="text-[9px] font-bold uppercase tracking-wider text-indigo-700 bg-indigo-50 px-2 py-0.5 rounded-full font-mono">
                  Type 01 / Outbound
                </span>
                <span className="text-slate-300"><Icons.ChevronRight /></span>
              </div>
              <p className="text-sm font-bold text-slate-900 mt-1.5">External Delivery Note</p>
              <p className="text-[11px] text-slate-500 mt-0.5 leading-snug">
                Outbound shipment to clients and vendors. Links to invoice on record.
              </p>
            </button>

            <button
              type="button"
              onClick={() => handleSelectGatewayType('internal')}
              className="w-full p-3.5 rounded-2xl border border-slate-200 hover:border-amber-600 hover:bg-amber-50/50 text-left transition active:scale-[0.98]"
            >
              <div className="flex items-center justify-between">
                <span className="text-[9px] font-bold uppercase tracking-wider text-amber-800 bg-amber-50 px-2 py-0.5 rounded-full font-mono">
                  Type 02 / Internal
                </span>
                <span className="text-slate-300"><Icons.ChevronRight /></span>
              </div>
              <p className="text-sm font-bold text-slate-900 mt-1.5">Internal Transfer Note</p>
              <p className="text-[11px] text-slate-500 mt-0.5 leading-snug">
                Stock movement between depots, workshops, and service centers.
              </p>
            </button>
          </div>

          <div className="flex items-center gap-3 py-1">
            <div className="flex-1 h-px bg-slate-200"></div>
            <span className="text-[9px] font-mono font-bold uppercase tracking-widest text-slate-400">or download blank</span>
            <div className="flex-1 h-px bg-slate-200"></div>
          </div>

          <div className="grid grid-cols-2 gap-2">
            <button
              type="button"
              onClick={() => showToast('Blank External PDF generated')}
              className="p-2.5 rounded-xl border border-slate-200 hover:bg-slate-50 text-left active:scale-95 transition"
            >
              <span className="text-[8px] font-mono font-bold uppercase text-slate-400 block">Blank Template</span>
              <span className="text-xs font-bold text-slate-700">External (PDF)</span>
            </button>
            <button
              type="button"
              onClick={() => showToast('Blank Internal PDF generated')}
              className="p-2.5 rounded-xl border border-slate-200 hover:bg-slate-50 text-left active:scale-95 transition"
            >
              <span className="text-[8px] font-mono font-bold uppercase text-slate-400 block">Blank Template</span>
              <span className="text-xs font-bold text-slate-700">Internal (PDF)</span>

          <div className="p-3 bg-slate-50 rounded-xl border border-slate-200/80 text-[11px] text-slate-600 leading-tight">

  // ==========================================

                    type="button"
                    onClick={() => handleUploadClick('sender')}
                    className="h-8 px-2.5 rounded-lg border border-slate-200 bg-white font-bold text-[11px] text-slate-700 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Upload /> Upload
                  </button>
                  <button
                    type="button"
                    onClick={() => setActiveDrawRole('sender')}
                    className="h-8 px-2.5 rounded-lg border border-slate-200 bg-white font-bold text-[11px] text-slate-700 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Draw /> Draw
                  </button>
                  <button
                    type="button"
                    onClick={() => setSignatorySheetRole('sender')}
                    className="h-8 px-3 rounded-lg border border-indigo-200 bg-indigo-50 font-bold text-[11px] text-indigo-700 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Person /> Pick
                  </button>
                </div>
              </div>
            )}
          </div>

          {/* Collected By Signature Card */}
          <div className="border border-slate-200 rounded-xl overflow-hidden bg-slate-50/50">
            <div className="p-3 border-b border-slate-200/80 flex items-center justify-between bg-white">
              <span className="font-bold text-xs text-slate-800">Collected By</span>
              <div className="flex items-center gap-2">
                <span className={`px-2 py-0.5 rounded-full text-[9px] font-black uppercase border ${
                  signatures.receiver.captured
                    ? 'bg-emerald-50 text-emerald-700 border-emerald-200'
                    : 'bg-slate-100 text-slate-400 border-slate-200'
                }`}>
                  {signatures.receiver.captured ? 'Captured' : 'Empty'}
                </span>
                <button
                  type="button"
                  onClick={() => setSignatures(prev => ({
                    ...prev,
                    receiver: { ...prev.receiver, visible: !prev.receiver.visible }
                  }))}
                  className="p-1 rounded text-slate-400 hover:text-slate-600"
                >
                  {signatures.receiver.visible ? <Icons.Eye /> : <Icons.EyeOff />}
                </button>
              </div>
            </div>

            {signatures.receiver.visible && (
              <div className="p-3 space-y-2">
                {signatures.receiver.captured && signatures.receiver.url ? (
                  <div className="h-20 rounded-lg border border-slate-200 bg-white flex items-center justify-center p-2 relative">
                    <img src={signatures.receiver.url} alt="Receiver Sign" className="max-h-full object-contain" />
                    <button
                      type="button"
                      onClick={() => setSignatures(prev => ({
                        ...prev,
                        receiver: { ...prev.receiver, captured: false, url: null }
                      }))}
                      className="absolute top-1.5 right-1.5 w-6 h-6 rounded-md bg-rose-50 text-rose-600 border border-rose-200 flex items-center justify-center"
                      title="Clear"
                    >
                      <Icons.Close />
                    </button>
                  </div>
                ) : (
                  <div className="h-16 rounded-lg border border-dashed border-slate-200 bg-white flex items-center justify-center text-[11px] text-slate-400 font-medium">
                    No signature captured yet
                  </div>
                )}

                <div className="flex items-center gap-1.5 flex-wrap">
                  <button
                    type="button"
                    onClick={() => handleUploadClick('receiver')}
                    className="h-8 px-2.5 rounded-lg border border-slate-200 bg-white font-bold text-[11px] text-slate-700 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Upload /> Upload
                  </button>
                  <button
                    type="button"
                    onClick={() => setActiveDrawRole('receiver')}
                    className="h-8 px-2.5 rounded-lg border border-slate-200 bg-white font-bold text-[11px] text-slate-700 flex items-center gap-1 active:scale-95 transition"
                  >
                    <Icons.Draw /> Draw
                  </button>
                </div>
              </div>
            )}
          </div>
        </section>

        {/* --- SECTION 6: NOTES (COLLAPSIBLE) --- */}
        <section className="bg-white rounded-2xl border border-slate-200 overflow-hidden shadow-sm">
          <button
            type="button"
            onClick={() => setNotesOpen(!notesOpen)}
            className="w-full p-3.5 flex items-center justify-between text-left hover:bg-slate-50 transition"
          >
            <div className="flex items-center gap-2">
              <span className="w-2 h-2 rounded-full bg-indigo-600"></span>
              <span className="text-xs font-bold text-slate-800">{notesTitle || 'Notes'}</span>
            </div>
            <div className="text-slate-400">
              {notesOpen ? <Icons.ChevronUp /> : <Icons.ChevronDown />}
            </div>
          </button>

          {notesOpen && (
            <div className="p-3.5 pt-0 space-y-2.5 border-t border-slate-100">
              <div>
                <label className="block text-[9px] font-black uppercase tracking-wider text-slate-400 mb-1">
                  Section Title
                </label>
                <input
                  type="text"
                  value={notesTitle}
                  onChange={(e) => setNotesTitle(e.target.value)}
                  className="w-full h-9 px-2.5 rounded-xl border border-slate-200 bg-slate-50 text-xs font-bold text-slate-800 focus:bg-white outline-none"
                />
              </div>
              <div>
                <label className="block text-[9px] font-black uppercase tracking-wider text-slate-400 mb-1">
                  Notes / Handling Remarks
                </label>
                <textarea
                  rows={3}
                  value={notesText}
                  onChange={(e) => setNotesText(e.target.value)}
                  placeholder="Security seals, special delivery instructions..."
                  className="w-full p-2.5 text-xs rounded-xl border border-slate-200 bg-slate-50 focus:bg-white outline-none resize-none font-medium"
                />
              </div>
            </div>
          )}
        </section>

        {/* --- SECTION 7: TERMS & CONDITIONS (CONDITIONAL COLLAPSIBLE) --- */}
        {showTerms && (
          <section className="bg-white rounded-2xl border border-slate-200 overflow-hidden shadow-sm">
            <button
              type="button"
              onClick={() => setTermsOpen(!termsOpen)}
              className="w-full p-3.5 flex items-center justify-between text-left hover:bg-slate-50 transition"
            >
              <div className="flex items-center gap-2">
                <span className="w-2 h-2 rounded-full bg-violet-600"></span>
                <span className="text-xs font-bold text-slate-800">Terms & Conditions</span>
              </div>
              <div className="text-slate-400">
                {termsOpen ? <Icons.ChevronUp /> : <Icons.ChevronDown />}
              </div>
            </button>

            {termsOpen && (
              <div className="p-3.5 pt-0 border-t border-slate-100">
                <textarea
                  rows={4}
                  value={termsText}
                  onChange={(e) => setTermsText(e.target.value)}
                  className="w-full p-2.5 text-[11px] font-mono leading-relaxed rounded-xl border border-slate-200 bg-slate-50 focus:bg-white outline-none resize-none"
                />
              </div>
            )}
          </section>
        )}

      </div>

      {/* --- M3 EXTENDED FLOATING ACTION BUTTON (FAB) --- */}
      {/* Replaces any wasted bottom bar while preserving maximum vertical viewable canvas */}
      <div className="fixed bottom-6 right-4 sm:right-8 z-40 flex items-center gap-2">
        <button
          type="button"
          onClick={() => handleSave('draft')}
          className="h-12 px-3.5 rounded-2xl bg-white text-slate-700 border border-slate-200 shadow-md font-bold text-xs active:scale-95 transition flex items-center gap-1.5"
          title="Save Draft"
        >
          Draft
        </button>

        <button
          type="button"
          onClick={() => handleSave('dispatched')}
          className="h-14 px-5 rounded-2xl bg-indigo-600 hover:bg-indigo-700 text-white font-black text-xs uppercase tracking-wider shadow-lg shadow-indigo-600/30 flex items-center gap-2 active:scale-95 transition"
        >
          <Icons.Save />
          <span>Save Waybill</span>
        </button>
      </div>

      {/* --- BOTTOM SHEET: CLIENT SELECTOR --- */}
      {clientOverlay && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-end">
          <div className="w-full max-w-md mx-auto bg-white rounded-t-3xl max-h-[82vh] flex flex-col p-4 space-y-3 animate-in slide-in-from-bottom duration-150">
            <div className="flex items-center justify-between pb-1 border-b border-slate-100">
              <span className="text-xs font-black uppercase tracking-wider text-slate-500">Select Client</span>
              <button onClick={() => setClientOverlay(false)} className="p-1 rounded-full text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <div className="relative">
              <input
                type="text"
                placeholder="Search clients..."
                value={clientQuery}
                onChange={(e) => setClientQuery(e.target.value)}
                className="w-full h-10 pl-9 pr-3 rounded-xl bg-slate-100 border-none text-xs font-medium focus:ring-2 focus:ring-indigo-600 outline-none"
                autoFocus
              />
              <div className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400">
                <Icons.Search />
              </div>
            </div>

            <div className="space-y-1.5 overflow-y-auto flex-1">
              {CLIENTS_DB.filter(c => c.name.toLowerCase().includes(clientQuery.toLowerCase())).map(c => (
                <div
                  key={c.id}
                  onClick={() => {
                    setClient(c);
                    setLocation(c.sub);
                    setClientOverlay(false);
                    showToast(`Client assigned: ${c.name}`);
                  }}
                  className="p-3 rounded-xl border border-slate-200 hover:bg-indigo-50/50 cursor-pointer active:scale-[0.99] transition flex items-center justify-between"
                >
                  <div>
                    <p className="font-bold text-xs text-slate-900">{c.name}</p>
                    <p className="text-[10px] text-slate-500 mt-0.5">{c.sub}</p>
                  </div>
                  <Icons.ChevronRight />
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* --- BOTTOM SHEET: INVOICE LINKER --- */}
      {invoiceOverlay && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-end">
          <div className="w-full max-w-md mx-auto bg-white rounded-t-3xl max-h-[82vh] flex flex-col p-4 space-y-3 animate-in slide-in-from-bottom duration-150">
            <div className="flex items-center justify-between pb-1 border-b border-slate-100">
              <div>
                <span className="text-xs font-black uppercase tracking-wider text-slate-500 block">Link Invoice</span>
                <span className="text-[10px] text-slate-400">Search for an invoice to link to this waybill</span>
              </div>
              <button onClick={() => setInvoiceOverlay(false)} className="p-1 rounded-full text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <div className="relative">
              <input
                type="text"
                placeholder="Search invoices by number, client, or PO..."
                value={invoiceQuery}
                onChange={(e) => setInvoiceQuery(e.target.value)}
                className="w-full h-10 pl-9 pr-3 rounded-xl bg-slate-100 border-none text-xs font-medium focus:ring-2 focus:ring-indigo-600 outline-none"
                autoFocus
              />
              <div className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400">
                <Icons.Search />
              </div>
            </div>

            <div className="space-y-1.5 overflow-y-auto flex-1">
              {INVOICES_DB.filter(inv =>
                inv.number.toLowerCase().includes(invoiceQuery.toLowerCase()) ||
                inv.client.toLowerCase().includes(invoiceQuery.toLowerCase()) ||
                inv.po.toLowerCase().includes(invoiceQuery.toLowerCase())
              ).map(inv => (
                <div
                  key={inv.id}
                  onClick={() => {
                    setLinkedInvoice(inv);
                    if (!poNumber) setPoNumber(inv.po);
                    setInvoiceOverlay(false);
                    showToast(`Linked ${inv.number}`);
                  }}
                  className="p-3 rounded-xl border border-slate-200 hover:bg-indigo-50/50 cursor-pointer active:scale-[0.99] transition flex items-center justify-between"
                >
                  <div>
                    <span className="font-mono font-bold text-xs text-indigo-900 block">{inv.number}</span>
                    <span className="text-[10px] text-slate-500">{inv.client} • {inv.po}</span>
                  </div>
                  <Icons.ChevronRight />
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* --- BOTTOM SHEET: SIGNATORY PICKER --- */}
      {signatorySheetRole && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-end">
          <div className="w-full max-w-md mx-auto bg-white rounded-t-3xl max-h-[82vh] flex flex-col p-4 space-y-3 animate-in slide-in-from-bottom duration-150">
            <div className="flex items-center justify-between pb-1 border-b border-slate-100">
              <div>
                <span className="text-xs font-black uppercase tracking-wider text-slate-500 block">Pick a signatory</span>
                <span className="text-[10px] text-slate-400">People who signed for you before. Tap to attach.</span>
              </div>
              <button onClick={() => setSignatorySheetRole(null)} className="p-1 rounded-full text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <div className="space-y-1.5 overflow-y-auto flex-1">
              {SIGNATORIES_DB.map(person => (
                <div
                  key={person.id}
                  onClick={() => handlePickSignatory(person)}
                  className="p-3 rounded-xl border border-slate-200 hover:bg-slate-50 cursor-pointer flex items-center justify-between active:scale-[0.99] transition"
                >
                  <div>
                    <p className="font-bold text-xs text-slate-900">{person.name}</p>
                    <p className="text-[10px] text-slate-500">{person.role}</p>
                  </div>
                  <span className="text-xs font-bold text-indigo-600">Select</span>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* --- MODAL: INLINE DRAW PAD --- */}
      {activeDrawRole && (
        <div className="fixed inset-0 z-50 bg-black/70 flex items-center justify-center p-3">
          <div className="w-full max-w-sm bg-white rounded-3xl p-4 space-y-3 animate-in zoom-in-95 duration-150">
            <div className="flex items-center justify-between pb-1 border-b border-slate-100">
              <span className="text-xs font-black uppercase tracking-wider text-slate-600">
                Draw Signature ({activeDrawRole === 'sender' ? 'Delivered By' : 'Collected By'})
              </span>
              <button onClick={() => setActiveDrawRole(null)} className="p-1 text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <div className="border border-slate-200 rounded-2xl bg-slate-50 overflow-hidden touch-none">
              <canvas
                ref={canvasRef}
                onMouseDown={startDraw}
                onMouseMove={moveDraw}
                onMouseUp={stopDraw}
                onTouchStart={startDraw}
                onTouchMove={moveDraw}
                onTouchEnd={stopDraw}
                className="w-full h-36 block cursor-crosshair bg-white"
              />
            </div>

            <div className="flex items-center justify-between pt-1">
              <button
                type="button"
                onClick={resetCanvas}
                className="px-3 py-1.5 rounded-xl border border-slate-200 text-[11px] font-bold text-slate-600 active:scale-95"
              >
                Reset
              </button>
              <div className="flex gap-1.5">
                <button
                  type="button"
                  onClick={() => setActiveDrawRole(null)}
                  className="px-3 py-1.5 text-[11px] font-bold text-slate-500"
                >
                  Cancel
                </button>
                <button
                  type="button"
                  onClick={saveDrawing}
                  className="px-4 py-1.5 rounded-xl bg-indigo-600 text-white text-[11px] font-bold active:scale-95 shadow"
                >
                  Save drawing
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* --- MODAL: COLUMN MANAGER & MORE SETTINGS --- */}
      {columnManagerOpen && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-3">
          <div className="w-full max-w-sm bg-white rounded-3xl p-5 space-y-4 animate-in zoom-in-95 duration-150">
            <div className="flex items-center justify-between pb-2 border-b border-slate-100">
              <span className="text-xs font-black uppercase tracking-wider text-slate-600">Table Settings</span>
              <button onClick={() => setColumnManagerOpen(false)} className="p-1 text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <div className="space-y-2">
              <span className="text-[10px] font-black uppercase text-slate-400 block">Standard Columns</span>
              <label className="flex items-center justify-between p-2.5 rounded-xl border border-slate-200">
                <span className="text-xs font-semibold">Make</span>
                <input
                  type="checkbox"
                  checked={columns.make}
                  onChange={(e) => setColumns({ ...columns, make: e.target.checked })}
                  className="w-4 h-4 rounded text-indigo-600"
                />
              </label>
              <label className="flex items-center justify-between p-2.5 rounded-xl border border-slate-200">
                <span className="text-xs font-semibold">Part No</span>
                <input
                  type="checkbox"
                  checked={columns.partNo}
                  onChange={(e) => setColumns({ ...columns, partNo: e.target.checked })}
                  className="w-4 h-4 rounded text-indigo-600"
                />
              </label>
              <label className="flex items-center justify-between p-2.5 rounded-xl border border-slate-200">
                <span className="text-xs font-semibold">Condition</span>
                <input
                  type="checkbox"
                  checked={columns.condition}
                  onChange={(e) => setColumns({ ...columns, condition: e.target.checked })}
                  className="w-4 h-4 rounded text-indigo-600"
                />
              </label>
            </div>

            {/* Custom Columns (Max 4) */}
            <div className="space-y-2 pt-1 border-t border-slate-100">
              <div className="flex items-center justify-between">
                <span className="text-[10px] font-black uppercase text-slate-400">Custom Columns ({customColumns.length}/4)</span>
                <button
                  type="button"
                  onClick={handleAddCustomColumn}
                  className="text-[11px] font-bold text-indigo-600 hover:text-indigo-800"
                >
                  + Add column
                </button>
              </div>

              {customColumns.map(col => (
                <div key={col.key} className="flex items-center justify-between p-2 rounded-xl bg-slate-50 border border-slate-200 text-xs font-semibold">
                  <span>{col.label}</span>
                  <button
                    type="button"
                    onClick={() => handleRemoveCustomColumn(col.key)}
                    className="p-1 text-slate-400 hover:text-rose-600"
                  >
                    <Icons.Close />
                  </button>
                </div>
              ))}
            </div>

            {/* More Settings Link */}
            <div className="pt-2 border-t border-slate-100">
              <button
                type="button"
                onClick={() => setMoreSettingsOpen(true)}
                className="w-full py-2.5 rounded-xl border border-slate-200 font-bold text-xs text-slate-700 hover:bg-slate-50"
              >
                More Settings (Terms & Conditions)
              </button>
            </div>

            <button
              type="button"
              onClick={() => setColumnManagerOpen(false)}
              className="w-full h-11 rounded-xl bg-indigo-600 text-white font-bold text-xs uppercase"
            >
              Done
            </button>
          </div>
        </div>
      )}

      {/* --- MODAL: MORE SETTINGS --- */}
      {moreSettingsOpen && (
        <div className="fixed inset-0 z-60 bg-black/60 flex items-center justify-center p-3">
          <div className="w-full max-w-xs bg-white rounded-3xl p-5 space-y-3 animate-in zoom-in-95 duration-150">
            <div className="flex items-center justify-between pb-2 border-b border-slate-100">
              <span className="text-xs font-black uppercase tracking-wider text-slate-600">More Settings</span>
              <button onClick={() => setMoreSettingsOpen(false)} className="p-1 text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <label className="flex items-center justify-between p-3 rounded-xl border border-slate-200">
              <span className="text-xs font-semibold">Show Terms & Conditions</span>
              <input
                type="checkbox"
                checked={showTerms}
                onChange={(e) => setShowTerms(e.target.checked)}
                className="w-4 h-4 rounded text-indigo-600"
              />
            </label>

            <button
              type="button"
              onClick={() => setMoreSettingsOpen(false)}
              className="w-full h-10 rounded-xl bg-indigo-600 text-white font-bold text-xs uppercase"
            >
              Done
            </button>
          </div>
        </div>
      )}

      {/* --- MODAL: IMPORT SHEET --- */}
      {importOverlay && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-3">
          <div className="w-full max-w-sm bg-white rounded-3xl p-5 space-y-3 animate-in zoom-in-95 duration-150">
            <div className="flex items-center justify-between pb-1 border-b border-slate-100">
              <div>
                <span className="text-xs font-black uppercase tracking-wider text-slate-600 block">Import Waybill</span>
                <span className="text-[10px] text-slate-400">Capture a paper waybill by pasting its JSON extraction.</span>
              </div>
              <button onClick={() => setImportOverlay(false)} className="p-1 text-slate-400">
                <Icons.Close />
              </button>
            </div>

            <textarea
              rows={6}
              value={importJson}
              onChange={(e) => setImportJson(e.target.value)}
              placeholder="Paste JSON here..."
              className="w-full p-2.5 text-xs font-mono rounded-xl border border-slate-200 bg-slate-50 focus:bg-white outline-none resize-none"
            />

            <div className="flex gap-2 pt-1">
              <button
                type="button"
                onClick={handleApplyImport}
                className="flex-1 h-10 rounded-xl border border-slate-200 bg-slate-50 font-bold text-xs text-slate-700 active:scale-95"
              >
                Preview
              </button>
              <button
                type="button"
                onClick={handleApplyImport}
                className="flex-1 h-10 rounded-xl bg-indigo-600 text-white font-bold text-xs active:scale-95"
              >
                Save
              </button>
            </div>
          </div>
        </div>
      )}

      {/* --- ALERT: CLEAR CONFIRMATION --- */}
      {clearAlertOpen && (
        <div className="fixed inset-0 z-60 bg-black/60 flex items-center justify-center p-3">
          <div className="w-full max-w-xs bg-white rounded-3xl p-5 space-y-3 animate-in zoom-in-95 duration-150">
            <div>
              <h3 className="text-sm font-black text-slate-900">Clear all items?</h3>
              <p className="text-xs text-slate-500 mt-1">
                This will remove all {items.length} line items from this document. This action cannot be undone.
              </p>
            </div>
            <div className="flex justify-end gap-2 pt-2">
              <button
                type="button"
                onClick={() => setClearAlertOpen(false)}
                className="px-3.5 py-2 text-xs font-bold text-slate-600"
              >
                Cancel
              </button>
              <button
                type="button"
                onClick={handleClearAll}
                className="px-4 py-2 rounded-xl bg-rose-600 text-white text-xs font-bold"
              >
                Clear All
              </button>
            </div>
          </div>
        </div>
      )}

      {/* --- TOAST NOTIFICATION --- */}
      {toastMessage && (
        <div className="fixed bottom-24 left-1/2 -translate-x-1/2 z-50 bg-slate-900/95 text-white text-xs font-medium px-4 py-2.5 rounded-2xl shadow-xl max-w-[85vw] truncate animate-in fade-in slide-in-from-bottom-2 duration-150">
          {toastMessage}
        </div>
      )}

    </div>
  );
}
